Código utilizado para crear el diagrama de Venn

Carga de paquetes y datos

library(ggplot2)
library(VennDiagram)

tabla <- read.csv("preferred_form_of_work_in_spain_in_2023.csv", fileEncoding = "UTF-8")
names(tabla) <- c("tipo", "porcentaje")

presencial <- tabla$porcentaje[tabla$tipo == "Presencial"]
hibrido <- tabla$porcentaje[tabla$tipo == "Híbrido"]
remoto  <- tabla$porcentaje[tabla$tipo == "Remoto"]

area1      <- presencial + hibrido
area2      <- remoto  + hibrido            
cross.area <- hibrido

Crear el diagrama

g <- draw.pairwise.venn(
  area1      = presencial + hibrido,
  area2      = remoto  + hibrido,
  cross.area = hibrido,
  category   = c("Presencial", "Remoto"),
  euler.d    = TRUE,
  scaled     = TRUE,
  ind        = FALSE,
  fill       = c("#0072B2", "#E69F00"),
  alpha      = 0.35,
  cex        = 1.5, cat.cex = 1.5,
  cat.pos    = c(-30, 30),
  cat.dist   = c(0.05, 0.05)
)

Pasar los círculos y textos en tablas

circulos <- data.frame()
for (i in seq_along(g)) {
  if (class(g[[i]])[1] == "polygon" && g[[i]]$gp$fill != "transparent") {
    circulos <- rbind(circulos, data.frame(x = as.numeric(g[[i]]$x), y = as.numeric(g[[i]]$y), grupo = i, color = g[[i]]$gp$fill))
  }
}

etiquetas <- data.frame()
for (i in seq_along(g)) {
  if (class(g[[i]])[1] == "text") {
    etiquetas <- rbind(etiquetas, data.frame(x = as.numeric(g[[i]]$x), y = as.numeric(g[[i]]$y), texto = ""))
  }
}
etiquetas$texto <- c(paste0(presencial, " %"), paste0(remoto, " %"), paste0("Híbrido\n", hibrido, " %"), "Presencial", "Remoto")

Crear el gráfico

p <- ggplot() +
  geom_polygon(data = circulos, aes(x, y, group = grupo, fill = color), alpha = 0.35) +
  geom_path(data = circulos, aes(x, y, group = grupo)) +
  geom_text(data = etiquetas, aes(x, y, label = texto), size = 6) +
  scale_fill_identity() +
  coord_fixed(ratio = 1) +
  labs(title = "Preferencia de formas de trabajo en España en 2023",
       caption = "Fuente de datos: Statista. Preferred form of work in Spain in 2023. https://www.statista.com/statistics/1450547/favorite-way-to-work-in-spain/") +
  theme_void() +
  theme(plot.title = element_text(face = "bold", size = 15, hjust = 0.5),
        plot.title.position = "plot", plot.caption.position = "plot",
        plot.caption = element_text(size = 8, hjust = 0))
